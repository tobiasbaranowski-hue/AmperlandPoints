## Stack
TanStack Start + Router (file-based routes in `src/routes`), React 19, Tailwind CSS 4 (Theme-Tokens in `src/styles.css`: `asphalt-*`, `ember-*`, `chalk`, Fonts Barlow / Barlow Condensed), lucide-react Icons, Netlify.
- `src/routes/admin.tsx` – mobile Admin-App (Bottom-Tabs, Bottom-Sheets, eigene Form-Bausteine unten in der Datei)
- `src/data/fixtures.ts` – Typen + alle Demo-Daten
- `src/data/store.ts` – `useCommunity()` / `setState()`; localStorage-basierter Demo-Speicher (useSyncExternalStore, Server-Snapshot = Fixtures)
- `src/components/` – SiteHeader/Logo, SectionTitle, Avatar
- `src/components/` – SiteHeader/Logo, SectionTitle, Avatar, Stars (Bewertung), TrackMap (GPX-Vorschau als SVG)
- `src/lib/activity.ts` – GPX-Parser, Foto-Verkleinerung, `photoSrc()`, Challenge-Rangliste, Punkte-Ranking
- `src/lib/format.ts` – `img()` (Image-CDN-URL), `formatDate()` (de-DE), `cn()`
- `public/img/*.png` – generierte Fotos, immer über `img(name, width)` einbinden
- `public/manifest.webmanifest`, `public/icon.svg` – PWA für die Admin-App (start_url `/admin`)
## Wichtige Entscheidungen
- Rollen (`gast`/`mitglied`/`admin`) sind aktuell nur Demo-Zustand im Store – keine echte Sicherheit. Wird mit Netlify Identity ersetzt.
- Persistenz gehört in Netlify Database (Drizzle, `db/schema.ts`); der Store ist die einzige Stelle, die ausgetauscht werden muss.
- Touren-Fotos werden im Demo als verkleinerte JPEG-Data-URLs im localStorage gespeichert (max. 4 pro Tour); `setState()` liefert `false`, wenn der Speicher voll ist.
- Challenge-km kommen ausschließlich aus hochgeladenen Touren im Challenge-Zeitraum; `member.km` ist nur der Gesamt-Tachostand.
- Keine Komponenten aus Route-Dateien exportieren (Code-Splitting).
- UI-Texte auf Deutsch.

import { cn } from '@/lib/format'

/** Schematische Vorschau einer GPX-Spur (ohne Kartenhintergrund). */
export function TrackMap({ track, className }: { track: Array<[number, number]>; className?: string }) {
  if (track.length < 2) return null
  const midLat = track.reduce((s, [lat]) => s + lat, 0) / track.length
  const k = Math.cos((midLat * Math.PI) / 180)
  const xs = track.map(([, lon]) => lon * k)
  const ys = track.map(([lat]) => -lat)
  const minX = Math.min(...xs), maxX = Math.max(...xs), minY = Math.min(...ys), maxY = Math.max(...ys)
  const span = Math.max(maxX - minX, maxY - minY) || 1
  const W = 200, H = 120, pad = 10
  const scale = Math.min((W - pad * 2) / (maxX - minX || span), (H - pad * 2) / (maxY - minY || span))
  const ox = (W - (maxX - minX) * scale) / 2, oy = (H - (maxY - minY) * scale) / 2
  const pts = xs.map((x, i) => [(x - minX) * scale + ox, (ys[i] - minY) * scale + oy])
  const d = pts.map(([x, y], i) => `${i ? 'L' : 'M'}${x.toFixed(1)} ${y.toFixed(1)}`).join('')
  const [sx, sy] = pts[0]
  const [ex, ey] = pts[pts.length - 1]
  return (
    <svg viewBox={`0 0 ${W} ${H}`} className={cn('bg-asphalt-800', className)} role="img" aria-label="Streckenverlauf">
      <defs>
        <pattern id="grid" width="20" height="20" patternUnits="userSpaceOnUse">
          <path d="M20 0H0V20" fill="none" stroke="currentColor" strokeOpacity=".06" />
        </pattern>
      </defs>
      <rect width={W} height={H} fill="url(#grid)" className="text-chalk" />
      <path d={d} fill="none" stroke="var(--color-ember-500)" strokeWidth="2.5" strokeLinejoin="round" strokeLinecap="round" />
      <circle cx={sx} cy={sy} r="4" fill="var(--color-chalk)" />
      <circle cx={ex} cy={ey} r="3" fill="var(--color-ember-400)" />
    </svg>
  )
}

// Erzeugt eine plausible Rundstrecke rund ums Amperland für die Demo-Touren.
function demoTrack(lat: number, lon: number, radius: number, wobble: number): Array<[number, number]> {
  const pts: Array<[number, number]> = []
  for (let i = 0; i <= 60; i++) {
    const a = (i / 60) * Math.PI * 2
    const r = radius * (1 + 0.25 * Math.sin(a * wobble))
    pts.push([+(lat + r * Math.sin(a)).toFixed(5), +(lon + r * 1.5 * Math.cos(a)).toFixed(5)])
  }
  return pts
}

export interface ParsedGpx {
  name?: string
  date?: string
  distanceKm: number
  track: Array<[number, number]>

const toRad = (d: number) => (d * Math.PI) / 180

function haversineKm([lat1, lon1]: [number, number], [lat2, lon2]: [number, number]) {
  const dLat = toRad(lat2 - lat1)
  const dLon = toRad(lon2 - lon1)
  const a = Math.sin(dLat / 2) ** 2 + Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLon / 2) ** 2
  return 6371 * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
}

/** Liest Track- oder Routenpunkte aus einer GPX-Datei und berechnet die Strecke. */
export function parseGpx(xml: string): ParsedGpx {
  const doc = new DOMParser().parseFromString(xml, 'application/xml')
  if (doc.querySelector('parsererror')) throw new Error('Keine gültige GPX-Datei')
  let nodes = Array.from(doc.getElementsByTagName('trkpt'))
  if (nodes.length === 0) nodes = Array.from(doc.getElementsByTagName('rtept'))
  const points = nodes
    .map((n) => [Number(n.getAttribute('lat')), Number(n.getAttribute('lon'))] as [number, number])
    .filter(([a, b]) => Number.isFinite(a) && Number.isFinite(b))
  if (points.length < 2) throw new Error('Die GPX-Datei enthält keine Strecke')

  let km = 0
  for (let i = 1; i < points.length; i++) km += haversineKm(points[i - 1], points[i])

  // Für Speicher und Vorschau auf höchstens ~200 Punkte ausdünnen
  const step = Math.max(1, Math.ceil(points.length / 200))
  const track = points
    .filter((_, i) => i % step === 0 || i === points.length - 1)
    .map(([a, b]) => [+a.toFixed(5), +b.toFixed(5)] as [number, number])

  const name = doc.querySelector('trk > name, rte > name, metadata > name')?.textContent?.trim() || undefined
  const time = doc.querySelector('trkpt > time, metadata > time')?.textContent?.slice(0, 10)
  return { name, date: time && /^\d{4}-\d{2}-\d{2}$/.test(time) ? time : undefined, distanceKm: Math.round(km), track }
}

/** Verkleinert ein Foto im Browser und gibt es als JPEG-Data-URL zurück (Demo-Speicher). */
export function compressImage(file: File, maxSize = 900, quality = 0.72): Promise<string> {
  return new Promise((resolve, reject) => {
    const url = URL.createObjectURL(file)
    const image = new Image()
    image.onload = () => {
      const scale = Math.min(1, maxSize / Math.max(image.width, image.height))
      const canvas = document.createElement('canvas')
      canvas.width = Math.round(image.width * scale)
      canvas.height = Math.round(image.height * scale)
      canvas.getContext('2d')?.drawImage(image, 0, 0, canvas.width, canvas.height)
      URL.revokeObjectURL(url)
      resolve(canvas.toDataURL('image/jpeg', quality))
    }
    image.onerror = () => {
      URL.revokeObjectURL(url)
      reject(new Error('Bild konnte nicht gelesen werden'))
    }
    image.src = url
  })
}

export function getMemberLevel(points: number): LevelInfo {
  const pts = Math.max(0, points)
  let current = LEVELS[0]
  for (const lvl of LEVELS) {
    if (pts >= lvl.minPoints) current = lvl
  }
  const currentIndex = LEVELS.findIndex((l) => l.level === current.level)
  const next = currentIndex < LEVELS.length - 1 ? LEVELS[currentIndex + 1] : null

  let progressPct = 100
  let pointsToNext = 0
  if (next) {
    const range = next.minPoints - current.minPoints
    const gained = pts - current.minPoints
    progressPct = Math.min(100, Math.max(0, Math.round((gained / range) * 100)))
    pointsToNext = Math.max(0, next.minPoints - pts)
  }

  return {
    level: current.level,
    title: current.title,
    desc: current.desc,
    color: current.color,
    currentPoints: pts,
    minPoints: current.minPoints,
    nextLevel: next,
    progressPct,
    pointsToNext,
  }
}
